140// method for the analytic solution
141vector<vector<double>> ProblemDefinition::ProblemDefinition::Analytic_Solution()
142{
143 // size of the vector result
144 int size_x = static_cast<int>(Get_thickness() / Get_dx()) + 1; // +1 to include endpoint
145 int size_t = static_cast<int>(Get_tMax() / Get_dt()) + 1; // +1 to include endpoint
146
147 // vector for the storage of the result
148 vector<vector<double>> analytic_result(size_x, vector<double>(size_t, 0.0));
149
150 // initialisation of variables
151 double x = 0.0;
152 double t = 0.0;
153 double sum = 0.0;
154
155 // first loop to store the result for each x position
156 for (int i=0; i<size_x; i++) {
157 t = 0.0; // initialisation of the t
158
159 // second loop to calculte the sum for each t
160 for (int j=0; j<size_t; j++) {
161 sum = 0.0; // initialisation of the sum
162
163 // third loop to calculate the sum
164 for (int k=1; k<=Get_N(); k++) {
165 sum += exp(- Get_D() * pow((k*M_PI / Get_thickness()), 2) * t) * ((1-pow(-1, k))/(k*M_PI)) * sin(k*M_PI*x/Get_thickness());
166 }
167
168 analytic_result[i][j] = Get_Tsur() + 2 * (Get_Tin() - Get_Tsur()) * sum;
169
170 t += Get_dt(); // increment the time
171 }
172
173 x += Get_dx(); // increment the position
174 }
175
176 return analytic_result;
177}
