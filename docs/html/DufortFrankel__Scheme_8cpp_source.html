27vector<vector<double>> DufortFrankel_Scheme::dfSolution() {
28 // size of the solution matrix
29 int num_space_points = static_cast<int>(this->problem.Get_thickness() / this->problem.Get_dx()) + 1;
30 int num_time_steps = static_cast<int>(this->problem.Get_tMax() / this->problem.Get_dt()) + 1;
31
32 // solution matrix with a size compose of time_index and space_index
33 vector<vector<double>> solution(num_space_points, vector<double>(num_time_steps, 0.0));
34
35 // vectors to store the different T
36 vector<double> T_prev(num_space_points); // T_n-1
37 vector<double> T_curr(num_space_points); // T_n
38 vector<double> T_next(num_space_points); // T_n+1
39
40 // Boundary conditions
41 T_curr[0] = this->problem.Get_Tsur();
42 T_curr[num_space_points-1] = this->problem.Get_Tsur();
43
44 // Initialisation of T_curr
45 for(int i=1; i<num_space_points-1; i++) {
46 T_curr[i] = this->problem.Get_Tin();
47 }
48
49 // Storage of the initial condition
50 for(int i=0; i<num_space_points; i++) {
51 solution[i][0] = T_curr[i];
52 }
53
54 // Initialisation of T_prev
55 T_prev = T_curr;
56
57 // Creation of the first time_step
58 vector<vector<double>> analytic = this->problem.Analytic_Solution(); // use of the analytical soluton
59 for(int i=0; i<num_space_points; i++) {
60 T_curr[i] = analytic[i][1];
61 solution[i][1] = T_curr[i];
62 }
63
64 // Loop to obtain all the other time steps
65 for (int t=2; t<num_time_steps; t++) { // time steps
66 // loop to obtain the result for each x
67 for (int i=1; i<num_space_points-1; i++) { // grid steps
68 T_next[i] = ((1.0-2.0*this->r)/(1.0+2.0*r)) * T_prev[i] + (2.0*this->r/(1.0+2.0*this->r)) * (T_curr[i-1] + T_curr[i+1]);
69 }
70
71 // Storage of the current step
72 for(int i=0; i<num_space_points; i++) {
73 solution[i][t] = T_next[i];
74 }
75
76 // Update of the T_xxx
77 T_prev = T_curr;
78 T_curr = T_next;
79 }
80
81 return solution;
82}
