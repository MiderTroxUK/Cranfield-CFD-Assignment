21CrankNicholson_Scheme::CrankNicholson_Scheme(const ProblemDefinition& problem)
22 // Call the base class 'Scheme' constructor
23 : Scheme(problem.Get_dx(), problem.Get_N(), problem.Get_dt(), problem.Get_D())
24{
25 // Calculate the 'r' parameter
26 this->r_ = (Get_D() * Get_dt()) / (Get_dx() * Get_dx());
27
28 // FIX: Calculate spatial nodes based on geometry, NOT time steps (N)
29 int num_space_points = static_cast<int>(problem.Get_thickness() / problem.Get_dx()) + 1;
30
31 // The tridiagonal system only applies to INTERNAL nodes
32 this->num_internal_nodes_ = num_space_points - 2;
33
34 // Resize vectors to fit the internal nodes
35 if (num_internal_nodes_ > 0) {
36 a_.resize(num_internal_nodes_);
37 b_.resize(num_internal_nodes_);
38 c_.resize(num_internal_nodes_);
39 d_.resize(num_internal_nodes_);
40 T_solution_.resize(num_internal_nodes_);
41
42 // Set up the (constant) matrix A coefficients
43 setup_thomas_coefficients();
44 } else {
45 // Handle the case where there are no internal nodes (e.g., N=2)
46 std::cerr << "ERROR: Number of spatial nodes must be > 2." << std::endl;
47 }
48}

50void CrankNicholson_Scheme::setup_thomas_coefficients() {
51 double val_a = -r_ / 2.0;
52 double val_b = 1.0 + r_;
53 double val_c = -r_ / 2.0;
54
55 for (int i = 0; i < num_internal_nodes_; ++i) {
56 a_[i] = val_a;
57 b_[i] = val_b;
58 c_[i] = val_c;
59 }

62// Sets up the RHS vector 'd_' based on the current temperature distribution and boundary conditions
63void CrankNicholson_Scheme::setup_rhs_vector(const std::vector<double>& T_current, double T_sur) {
64
65 // 1. Middle internal nodes (from i_int = 1 to N-4)
66 // (corresponding to physical nodes j=2 to N-3)
67 for (int i_int = 1; i_int < num_internal_nodes_ - 1; ++i_int) {
68 int j_phys = i_int + 1; // Corresponding physical node
69 d_[i_int] = (r_ / 2.0) * T_current[j_phys - 1] +
70 (1.0 - r_) * T_current[j_phys] +
71 (r_ / 2.0) * T_current[j_phys + 1];
72 }
73
74 // 2. Handle the first internal node (i_int = 0, j_phys = 1)
75 if (num_internal_nodes_ > 0) {
76 d_[0] = r_ * T_sur +
77 (1.0 - r_) * T_current[1] +
78 (r_ / 2.0) * T_current[2];
79 }
80
81 // 3. Handle the last internal node (i_int = N-3, j_phys = N-2)
82 if (num_internal_nodes_ > 1) {
83 int i_last = num_internal_nodes_ - 1;
84 int j_last = i_last + 1;
85
86 d_[i_last] = (r_ / 2.0) * T_current[j_last - 1] +
87 (1.0 - r_) * T_current[j_last] +
88 r_ * T_sur;
89 }
90}

93// Solves the tridiagonal system using the Thomas algorithm
94void CrankNicholson_Scheme::solve_thomas_algorithm() {
95 if (num_internal_nodes_ <= 0) return;
96
97 // Create temporary copies for the elimination pass
98 vector<double> c_prime = c_;
99 vector<double> d_prime = d_;
100
101 // 1. Forward Elimination pass
102 c_prime[0] = c_prime[0] / b_[0];
103 d_prime[0] = d_prime[0] / b_[0];
104
105 for (int i = 1; i < num_internal_nodes_; ++i) {
106 double m = b_[i] - a_[i] * c_prime[i - 1];
107 if (std::abs(m) < 1e-10) {
108 cerr << "ERROR: Thomas algorithm unstable (division by zero)." << endl;
109 return;
110 }
111 c_prime[i] = c_prime[i] / m;
112 d_prime[i] = (d_prime[i] - a_[i] * d_prime[i - 1]) / m;
113 }
114
115 // 2. Backward Substitution pass
116 T_solution_[num_internal_nodes_ - 1] = d_prime[num_internal_nodes_ - 1];
117
118 for (int i = num_internal_nodes_ - 2; i >= 0; --i) {
119 T_solution_[i] = d_prime[i] - c_prime[i] * T_solution_[i + 1];
120 }
121}

123// Performs one time step of the Crank-Nicholson scheme
124void CrankNicholson_Scheme::solve_step(std::vector<double>& T_current, double T_sur) {
125 if (num_internal_nodes_ <= 0) return;
126
127 setup_rhs_vector(T_current, T_sur);
128 solve_thomas_algorithm();
129
130 for (int i = 0; i < num_internal_nodes_; ++i) {
131 T_current[i + 1] = T_solution_[i];
132 }
133}

135vector<vector<double>> CrankNicholson_Scheme::Solve() {
136 int num_time_steps = static_cast<int>(Get_N()) + 1;
137 // Calculate spatial points based on thickness and dx
138 int num_space_points = static_cast<int>(31.0 / Get_dx()) + 1;
139
140 // Initialize solution matrix [space x time]
141 vector<vector<double>> solution(num_space_points, vector<double>(num_time_steps, 0.0));
142
143 // Current temperature profile
144 vector<double> T_curr(num_space_points);
145
146 // 1. Set Initial Conditions (t=0)
147 for (int i = 0; i < num_space_points; ++i) {
148 T_curr[i] = 38.0; // Tin
149 }
150
151 // Boundary Conditions
152 T_curr[0] = 149.0; // Tsur
153 T_curr[num_space_points - 1] = 149.0;
154
155 // Store IC into solution column 0
156 for (int i = 0; i < num_space_points; ++i) {
157 solution[i][0] = T_curr[i];
158 }
159
160 // 2. Time Loop
161 for (int t = 1; t < num_time_steps; ++t) {
162 solve_step(T_curr, 149.0);
163
164 for (int i = 0; i < num_space_points; ++i) {
165 solution[i][t] = T_curr[i];
166 }
167 }
168
169 return solution;
170}
