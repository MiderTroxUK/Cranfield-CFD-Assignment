21Laasonen_Scheme::Laasonen_Scheme(const ProblemDefinition& problem)
22 : Scheme(problem.Get_dx(), problem.Get_N(), problem.Get_dt(), problem.Get_D())
23{
24 // Calculate 'r'
25 this->r_ = (Get_D() * Get_dt()) / (Get_dx() * Get_dx());
26
27 // L = 31.0 cm, dx = 0.05 cm -> ~621 nodes

46// 1. Matrix Coefficients (Different from Crank-Nicholson)
47void Laasonen_Scheme::setup_thomas_coefficients() {
48 // Equation: -r*T(i-1) + (1 + 2r)*T(i) - r*T(i+1) = T_old(i)
49
50 double val_a = -r_;
51 double val_b = 1.0 + 2.0 * r_;
52 double val_c = -r_;
53
54 for (int i = 0; i < num_internal_nodes_; ++i) {
55 a_[i] = val_a;
56 b_[i] = val_b;
57 c_[i] = val_c;
58 }
59}

61// 2. RHS Vector (Different from Crank-Nicholson)
62void Laasonen_Scheme::setup_rhs_vector(const std::vector<double>& T_current, double T_sur) {
63
64 // General case for internal nodes: RHS is just the old temperature
65 for (int i_int = 0; i_int < num_internal_nodes_; ++i_int) {
66 int j_phys = i_int + 1;
67 d_[i_int] = T_current[j_phys];
68 }
69
70 // Boundary Conditions adjustment
71 // The term -r*T(i-1) moves to RHS and becomes +r*T_sur for the first node
72 if (num_internal_nodes_ > 0) {
73 d_[0] += r_ * T_sur;
74 }
75
76 // The term -r*T(i+1) moves to RHS and becomes +r*T_sur for the last node
77 if (num_internal_nodes_ > 1) {
78 d_[num_internal_nodes_ - 1] += r_ * T_sur;
79 }
80}

82// 3. Solver (Identical to Crank-Nicholson)
83void Laasonen_Scheme::solve_thomas_algorithm() {
84 if (num_internal_nodes_ <= 0) return;
85
86 vector<double> c_prime = c_;
87 vector<double> d_prime = d_;
88
89 // Forward Elimination
90 c_prime[0] = c_prime[0] / b_[0];
91 d_prime[0] = d_prime[0] / b_[0];
92
93 for (int i = 1; i < num_internal_nodes_; ++i) {
94 double m = b_[i] - a_[i] * c_prime[i - 1];
95 if (std::abs(m) < 1e-10) return; // Stability check
96
97 c_prime[i] = c_prime[i] / m;
98 d_prime[i] = (d_prime[i] - a_[i] * d_prime[i - 1]) / m;
99 }
100
101 // Backward Substitution
102 T_solution_[num_internal_nodes_ - 1] = d_prime[num_internal_nodes_ - 1];
103
104 for (int i = num_internal_nodes_ - 2; i >= 0; --i) {
105 T_solution_[i] = d_prime[i] - c_prime[i] * T_solution_[i + 1];
106 }
107}

109// 4. Main Step function
110void Laasonen_Scheme::solve_step(std::vector<double>& T_current, double T_sur) {
111
112 if (num_internal_nodes_ <= 0) return;
113
114 setup_rhs_vector(T_current, T_sur);
115 solve_thomas_algorithm();
116
117 for (int i = 0; i < num_internal_nodes_; ++i) {
118 T_current[i + 1] = T_solution_[i];
119 }
120}

122vector<vector<double>> Laasonen_Scheme::Solve() {
123 int num_time_steps = static_cast<int>(Get_N()) + 1;
124 // Calculate spatial points based on thickness and dx
125 // Using L = 31 cm (Hardcoded here or ideally passed via a getter if added to Scheme)
126 // Since we don't have Get_L() in Scheme, we assume the calculation below is correct relative to the problem
127 int num_space_points = static_cast<int>(31.0 / Get_dx()) + 1;
128
129 // Initialize solution matrix [space x time]
130 vector<vector<double>> solution(num_space_points, vector<double>(num_time_steps, 0.0));
131
132 // Current temperature profile
133 vector<double> T_curr(num_space_points);
134
135 // 1. Set Initial Conditions (t=0)
136 for (int i = 0; i < num_space_points; ++i) {
137 T_curr[i] = 38.0; // Tin
138 }
139
140 // Boundary Conditions
141 T_curr[0] = 149.0; // Tsur
142 T_curr[num_space_points - 1] = 149.0;
143
144 // Store IC into solution column 0
145 for (int i = 0; i < num_space_points; ++i) {
146 solution[i][0] = T_curr[i];
147 }
148
149 // 2. Time Loop
150 for (int t = 1; t < num_time_steps; ++t) {
151 // Perform one implicit Laasonen step (updates T_curr)
152 solve_step(T_curr, 149.0);
153
154 // Store result at time t
155 for (int i = 0; i < num_space_points; ++i) {
156 solution[i][t] = T_curr[i];
157 }
158 }
159
160 return solution;
161}
