28vector<vector<double>> Richardson_Scheme::richardsonSolution(vector<vector<double>> laasonen) {
29 // size of the solution matric
30 int num_time_steps = static_cast<int>(this->problem.Get_tMax() / this->problem.Get_dt()) + 1;
31 int num_space_points = static_cast<int>(this->problem.Get_thickness() / this->problem.Get_dx()) +1;
32
33 // solution matrix with a size compose of time_index and space_index
34 vector<vector<double>> solution(num_space_points, vector<double>(num_time_steps, 0.0));
35
36 // vectors to store the different T
37 vector<double> T_prev(num_space_points); // T_n-1
38 vector<double> T_curr(num_space_points); // T_n
39 vector<double> T_next(num_space_points); // T_n+1
40
41 // Boundary conditions
42 T_curr[0] = this->problem.Get_Tsur();
43 T_curr[num_space_points-1] = this->problem.Get_Tsur();
44
45 // Initialisation of T_curr
46 for(int i=1; i<num_space_points-1; i++) {
47 T_curr[i] = this->problem.Get_Tin();
48 }
49
50 // Storage of the initial condition
51 for(int i=0; i<num_space_points; i++) {
52 solution[i][0] = T_curr[i];
53 }
54
55 // Initialisation of T_prev
56 T_prev = T_curr;
57
58 // Creation of the first time_step
59 for(int i=0; i<num_space_points; i++) {
60 T_curr[i] = laasonen[i][1];
61 solution[i][1] = T_curr[i];
62 }
63
64 // Loop to obtain all the other time steps
65 for (int t=2; t<num_time_steps; t++) { // time steps
66 // loop to obtain the result for each x
67 for (int i=1; i<num_space_points-1; i++) { // grid steps
68 T_next[i] = T_prev[i] + 2.0 * this->r * (T_curr[i+1] - 2.0 * T_curr[i] + T_curr[i-1]);
69 }
70
71 // Storage of the current step
72 for(int i=0; i<num_space_points; i++) {
73 solution[i][t] = T_next[i];
74 }
75
76 // Update of the T_xxx
77 T_prev = T_curr;
78 T_curr = T_next;
79 }
80
81 return solution;
82}
